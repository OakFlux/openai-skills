            first = post_query(stock_value, '', keyword, 1, 100)
            local.extend(first.get('announcements') or [])
            total_pages = int(first.get('totalpages') or first.get('totalPages') or 1)
            for page in range(2, min(total_pages, 5) + 1):
                result = post_query(stock_value, '', keyword, page, 100)
                local.extend(result.get('announcements') or [])
        if local:
            all_announcements = local
            chosen_stock = stock_value
            break
    except Exception as exc:
        print('QUERY_VARIANT_ERR', stock_value, repr(exc), flush=True)

if not all_announcements:
    raise RuntimeError('CNInfo returned no announcements for 依依股份 001206')

unique: dict[str, dict[str, Any]] = {}
for item in all_announcements:
    code_blob = str(item.get('secCode') or item.get('stockCode') or '')
    name_blob = str(item.get('secName') or item.get('stockName') or '')
    if code_blob and CODE not in code_blob:
        continue
    if name_blob and COMPANY not in name_blob and FULL_COMPANY not in name_blob:
        continue
    key = str(item.get('announcementId') or item.get('adjunctUrl') or json.dumps(item, ensure_ascii=False, sort_keys=True))
    unique[key] = item
announcements = list(unique.values())
if not announcements:
    raise RuntimeError('Filtered CNInfo announcements are empty')
print('CHOSEN_STOCK', chosen_stock, 'ANNOUNCEMENT_COUNT', len(announcements), flush=True)

    title = str(item.get('announcementTitle') or item.get('title') or '')
    title = re.sub(r'<[^>]+>', '', title).replace('&nbsp;', ' ')
