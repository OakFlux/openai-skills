        continue
    exact = int(bool(re.fullmatch(
        rf'(?:依依股份|天津市依依卫生用品股份有限公司)?[:：]?{year}年年度报告(?:（修订版）|\(修订版\)|（更新后）|\(更新后\))?',
        title,
    )))
    revised = int(any(term in title for term in ['修订版', '更新后']))
    annual_candidates.setdefault(year, []).append(((exact, revised, pub_date(item)), item))

for year, candidates in annual_candidates.items():
    candidates.sort(key=lambda entry: entry[0], reverse=True)
    annual_by_year[year] = candidates[0][1]

if not annual_by_year:
    raise RuntimeError('No complete annual reports found')

# Final IPO prospectus: exclude summary, application draft, and ancillary notices.
prospectus_candidates: list[tuple[tuple[int, int, int, str], dict[str, Any]]] = []
for item in announcements:
    title = plain_title(item)
    if '招股说明书' not in title:
        continue
    if any(term in title for term in ['摘要', '申报稿', '提示性公告', '发行公告', '上市公告', '问询', '反馈']):
        continue
    score_exact = int('首次公开发行股票招股说明书' in title)
    score_final = int(not any(term in title for term in ['修订稿', '预披露']))
    score_company = int(COMPANY in title or FULL_COMPANY in title or title.startswith('首次公开发行'))
    prospectus_candidates.append(((score_exact, score_final, score_company, pub_date(item)), item))

if not prospectus_candidates:
    related = [plain_title(item) for item in announcements if '招股说明书' in plain_title(item)]
    raise RuntimeError(f'No final prospectus found; related={related}')
