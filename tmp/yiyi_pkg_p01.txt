def walk(obj: Any) -> None:
    if isinstance(obj, dict):
        blob = json.dumps(obj, ensure_ascii=False)
        code = str(obj.get('code') or obj.get('secCode') or obj.get('stockCode') or obj.get('SECCODE') or '')
        name = str(obj.get('zwjc') or obj.get('secName') or obj.get('stockName') or obj.get('name') or '')
        org_id = obj.get('orgId') or obj.get('orgID') or obj.get('orgid')
        if CODE in (code + blob) and (COMPANY in (name + blob) or FULL_COMPANY in blob) and org_id:
            stock_candidates.append(f'{CODE},{org_id}')
        for value in obj.values():
            walk(value)
    elif isinstance(obj, list):
        for value in obj:
            walk(value)
